// Hàm kiểm tra email
function checkEmail(email) {

    // Xóa khoảng trắng dư
    email = email.trim();

    /*
    Lỗi 1:
    includes() chỉ kiểm tra chuỗi có chứa ký tự.
    Nên dùng endsWith() để kiểm tra đúng đuôi email.
    */

    if (!email.endsWith("@dhcd.edu.vn")) {

        alert("Email không hợp lệ");

        return false;

    } else {

        console.log("Email đúng định dạng");

        return true;
    }
}
